   230→  // Handle Assessment Completion from Player
   231→  async function handleLessonAssessment(lessonId: string, answerIdx: number): Promise<boolean> {
   232→    const lesson = lessons.find(l => l.id === lessonId);
   233→    if (!lesson) return false;
   234→
   235→    if (user) {
   236→      try {
   237→        const res = await fetch("/api/assessments", {
   238→          method: "POST",
   239→          headers: { "Content-Type": "application/json" },
   240→          body: JSON.stringify({ lessonId, answer: answerIdx }),
   241→        });
   242→        const data = await res.json();
   243→        if (res.ok && data.correct) {
   244→          setCompleted(prev => [...new Set([...prev, lessonId])]);
   245→          if (data.xpGained) setExperiencePoints(xp => xp + data.xpGained);
   246→          return true;
   247→        }
   248→        return false;
   249→      } catch {
   250→        return false;
   251→      }
   252→    } else {
   253→      // Guest local execution
   254→      const isCorrect = answerIdx === lesson.answer;
   255→      if (isCorrect) {
   256→        const updated = [...new Set([...completed, lessonId])];
   257→        setCompleted(updated);
   258→        setExperiencePoints(xp => xp + 50);
   259→        try {
   260→          localStorage.setItem("atlas_demo_completed", JSON.stringify(updated));
   261→        } catch {}
   262→      }
   263→      return isCorrect;
   264→    }
   265→  }

   284→        setCoachHistory(prev => [

   290→          ...prev,

   295→      setCoachHistory(prev => [
   296→        ...prev,
   297→        { question: query, answer: "Connection error. Please try again." }
   298→      ]);
   299→    }

   305→    e.preventDefault();

   310→    try {

   312→        method: "POST",
   313→        headers: { "Content-Type": "application/json" },
   314→        body: JSON.stringify({
   315→          action: authMode,
   316→          name: form.get("name"),
   317→          email: form.get("email"),
   318→          password: form.get("password"),
   319→        }),
   320→      });
   321→      const data = await res.json();
   322→      if (!res.ok) {
   323→        setAuthError(data.error || "Authentication failed.");
   324→      } else {
   325→        setAuthOpen(false);
   326→        window.location.href = "/";
   327→      }
   328→    } catch {
   329→      setAuthError("Server unavailable. Please try again.");
